GENS = [ux**2 + uy**2 + uz**2 - 1, vx**2 + vy**2 + vz**2 - 1]
SYMS = [ux, uy, uz, vx, vy, vz]

def zero_mod_unit(M):
    # entrywise polynomial reduction modulo the ideal <u.u-1, v.v-1>
    # (w1, w2 carried as extra polynomial generators absent from the ideal)
    for e in M:
        num, den = sp.fraction(sp.cancel(sp.expand(e)))
        gens = SYMS + ([w1, w2] if (num.has(w1) or num.has(w2)) else [])
        r = sp.reduced(sp.Poly(sp.expand(num), *gens, domain='QQ_I').as_expr(),
                       GENS, *gens)[1]
        if sp.simplify(r) != 0:
            return False
    return True

ok = lambda name, cond: print(f"{name}: {'PASS' if cond else 'FAIL'}")

Au = ux*al[0] + uy*al[1] + uz*al[2]    # alpha.u  (u = k_hat)
Av = vx*al[0] + vy*al[1] + vz*al[2]    # alpha.v  (v = k_hat')
Pp = lambda A: (I4 + A)/2              # P_+ at unit vector argument
Pm = lambda A: (I4 - A)/2              # P_-

ok("J3  (a.k')S - S(a.k) = (a.(k+k'))S         ",

# J4: band resolution of the commutator vertex, all four sign pairs:
#     P_{s'}(v) (alpha.(k+k')) S P_s(u) = (s' w2 - s w1) P_{s'}(v) P_{-s}(u) S
V4 = w1*Au + w2*Av
res = []
for sp_, Ppr in ((1, Pp), (-1, Pm)):
    for s_, Psl in ((1, Pp), (-1, Pm)):
        Pneg = Pm if s_ == 1 else Pp   # P_{-s}
        lhs = Ppr(Av) * V4 * S * Psl(Au)
        rhs = (sp_*w2 - s_*w1) * Ppr(Av) * Pneg(Au) * S
        res.append(zero_mod_unit(sp.expand(lhs - rhs)))
ok("J4  P_s'(v)(a.(k+k'))S P_s(u)=(s'|k'|-s|k|)P_s'(v)P_-s(u)S (4 sign pairs)", all(res))

# J5: the projector-mismatch COINCIDENCE factor (exact, one order):

   and zero_mod_unit(sp.expand(Pm(Av)*Pp(Au) - sp.Rational(1, 2)*Pm(Av)*Duv)))

# J6: the D(t)-kernel time factor: the commutator's degree-(+1) prefactor cancels EXACTLY:
#     i*omega * int_0^t e^{i omega s} ds = e^{i omega t} - 1  (omega != 0; the omega = 0

   sp.simplify(sp.I*omn*sp.integrate(sp.exp(sp.I*omn*s_), (s_, 0, t))
               - (sp.exp(sp.I*omn*t) - 1)) == 0
   and sp.simplify(branch0.subs(om, 0)) == 0)

# J7: the equal-time weight-sum collapse is SPINOR-TRIVIAL with the c-profile:

   and sp.simplify(g.subs(rho, 0)) == 1
   and sp.simplify(g.subs(rho, sp.Rational(1, 2))) == 0
   and sp.simplify(sp.diff(g, rho).subs(rho, sp.Rational(1, 2))) == 0)
