    piv = {}
    r = 0
    for row in rows:
        cur = {c: v % p for c, v in row.items() if v % p}

            if c in piv:
                f = cur[c]
                base = piv[c]
                nxt = dict(cur)

                r += 1
                cur = {}
    return r
